#### 3.2 Request Size Limits Middleware

**New File:** `src/server/api/middleware/request-limits.ts`

```typescript
import type { MiddlewareHandler } from "hono";
import { HTTPException } from "hono/http-exception";
import type { AppBindings } from "../types";

const MAX_JSON_SIZE = 10 * 1024 * 1024; // 10MB
const MAX_HEADER_SIZE = 8 * 1024; // 8KB

export const requestLimits: MiddlewareHandler<AppBindings> = async (c, next) => {
  const contentLength = c.req.header("content-length");
  if (contentLength && parseInt(contentLength) > MAX_JSON_SIZE) {
    throw new HTTPException(413, {
      message: `Request too large. Max ${MAX_JSON_SIZE / 1024 / 1024}MB`,
    });
  }

  for (const [key, value] of Object.entries(c.req.header())) {
    if (value && value.length > MAX_HEADER_SIZE) {
      throw new HTTPException(431, { message: `Header ${key} too large` });
    }
  }

  await next();
};
```

```typescript
interface DailyCosts {
  transcription_usd: number;
  openai_usd: number;
  total_usd: number;
}

export async function getDailyCosts(date = new Date()): Promise<DailyCosts> {
  const sql = getSql();
  const dateStr = date.toISOString().slice(0, 10); // YYYY-MM-DD

  const [costs] = await sql<[DailyCosts]>`
    SELECT 
      COALESCE(SUM(CASE WHEN usage_type = 'transcription' THEN amount * 0.10 / 3600 ELSE 0 END), 0) as transcription_usd,
      COALESCE(SUM(CASE WHEN usage_type = 'ai_query' THEN amount * 0.02 ELSE 0 END), 0) as openai_usd,
      COALESCE(SUM(total_cost_usd), 0) as total_usd
    FROM usage_logs
    WHERE DATE(created_at) = ${dateStr}
  `;

  return costs;
}

export async function checkCostBudget(): Promise<void> {
  const today = await getDailyCosts();
  const budget = parseFloat(process.env.DAILY_COST_BUDGET_USD || "1000");

  if (today.total_usd > budget * 1.5) {

      type: "cost_spike",
      details: {
        daily_cost: today.total_usd,
        budget,
        ratio: today.total_usd / budget,
        message: "CRITICAL: Daily costs 150% over budget",
      },
    });
  }
}
```

#### 4.4 Error Retry Wrapper

**New File:** `src/server/lib/retry.ts`

**Create:**

```typescript
export async function withRetry<T>(
  fn: () => Promise<T>,
  options: {
    maxAttempts?: number;
    backoffMs?: number;
    onRetry?: (attempt: number, error: Error) => void;
  } = {},
): Promise<T> {
  const { maxAttempts = 3, backoffMs = 1000, onRetry } = options;

  for (let attempt = 1; attempt <= maxAttempts; attempt++) {
    try {
      return await fn();
    } catch (error) {
      if (attempt === maxAttempts) throw error;

      const delay = backoffMs * Math.pow(2, attempt - 1);
      console.warn(`Attempt ${attempt} failed, retrying in ${delay}ms...`);

      if (onRetry) onRetry(attempt, error as Error);
      await new Promise((resolve) => setTimeout(resolve, delay));
    }
  }

  throw new Error("Unreachable");
}
```

**Use in:** `src/server/workers/processing.ts`

```typescript
import { getRedis } from "../services/redis";

export async function withCache<T>(
  key: string,
  ttlSeconds: number,
  fn: () => Promise<T>,
): Promise<T> {
  const redis = getRedis();

  try {
    const cached = await redis.get(key);
    if (cached) {
      return JSON.parse(cached) as T;
    }
  } catch (err) {
    console.warn("[Cache] Read failed, falling back to source:", err);
  }

  const result = await fn();

  try {
    await redis.setex(key, ttlSeconds, JSON.stringify(result));
  } catch (err) {
    console.warn("[Cache] Write failed:", err);
  }

  return result;
}
```

#### 6.6 Webhook Notifications (Optional)

**New File:** `src/server/services/webhooks.ts`

**Create:**

```typescript
export async function sendWebhook(
  url: string,
  event: string,
  payload: Record<string, any>,
): Promise<void> {
  try {
    const response = await fetch(url, {
      method: "POST",
      headers: {
        "content-type": "application/json",
        "user-agent": "EchoBrief-Webhook/1.0",
      },
      body: JSON.stringify({
        event,
        timestamp: new Date().toISOString(),
        data: payload,
      }),
    });

    if (!response.ok) {
      console.warn(`[Webhook] Failed: ${url} - ${response.status}`);
    }
  } catch (error) {
    console.error(`[Webhook] Error: ${url}`, error);
  }
}
```

**Use in:** `src/server/workers/processing.ts`

```typescript
import { sendWebhook } from "../services/webhooks";

// After job completes
const meeting = await getMeeting(job.data.meeting_id);
if (meeting.webhook_url) {
  await sendWebhook(meeting.webhook_url, "meeting.processed", {
    meeting_id: meeting.id,
    status: "completed",
  });
}
```

**Why:** Real-time notifications for external integrations.

#### 6.7 Worker Monitoring Dashboard

**New File:** `src/server/api/routes/admin-workers.ts`

**Create:**

```typescript
import { Hono } from "hono";
import { requireAdmin } from "../middleware/auth";
import { getProcessingQueue } from "../../services/queue";

const app = new Hono();
app.use("*", requireAdmin);

app.get("/workers/stats", async (c) => {
  const queue = getProcessingQueue();

  const [waiting, active, completed, failed] = await Promise.all([
    queue.getWaitingCount(),
    queue.getActiveCount(),
    queue.getCompletedCount(),
    queue.getFailedCount(),
  ]);

  return c.json({
    queue: {
      waiting,
      active,
      completed,
      failed,
    },
    timestamp: Date.now(),
  });
});

export default app;
```
