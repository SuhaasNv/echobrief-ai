export async function exchangeNotionToken(code: string, redirectUri: string): Promise<TokenResult> {
  const response = await fetch("https://api.notion.com/v1/oauth/token", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      grant_type: "authorization_code",
      code,
      redirect_uri: redirectUri,
    }),
    auth: { username: NOTION_CLIENT_ID, password: NOTION_CLIENT_SECRET },
  });
  const data = await response.json();
  return {
    access_token: data.access_token,
    refresh_token: null, // Notion doesn't use refresh tokens
    expires_at: null,
    workspace_name: data.workspace_name,
  };
}

**Create:** `src/lib/hooks.ts`

```typescript
export function useDebouncedValue<T>(value: T, delay: number): T {
  const [debouncedValue, setDebouncedValue] = useState(value);

  useEffect(() => {
    const handler = setTimeout(() => setDebouncedValue(value), delay);
    return () => clearTimeout(handler);
  }, [value, delay]);

  return debouncedValue;
}
```

const toggleMutation = useMutation({
  mutationFn: (id: string) => patchActionItem(id, { completed: !item.completed }),
  onMutate: async (id) => {
    // Optimistically update UI
    await queryClient.cancelQueries({ queryKey: ["action-items"] });
    const prev = queryClient.getQueryData(["action-items"]);
