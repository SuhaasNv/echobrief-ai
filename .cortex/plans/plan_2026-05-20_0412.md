#### 1.2 Add Input Sanitization for Prompt Injection

**File:** `src/server/lib/prompts.ts` (new utility functions)

**Problem:** User transcripts could contain adversarial instructions:

**Solution - Multi-Layer Defense:**

**Layer 1: Input Sanitization**

```typescript
// NEW FILE: src/server/lib/sanitization.ts
import { z } from "zod";

export const SUSPICIOUS_PATTERNS = [
  /ignore\s+(all\s+)?(previous|above|prior)\s+instructions/i,
  /you\s+are\s+now\s+a\s+/i,
  /disregard\s+(all\s+)?(previous|above|prior)/i,
  /forget\s+(everything|all)\s+(you|that)/i,
  /new\s+instructions:/i,
  /<system>|<\/system>|<prompt>|<\/prompt>/i, // XML tag injection
  /\{\{.*system.*\}\}|%%.*system.*%%/i, // Template injection
];

export function sanitizeTranscript(text: string): string {
  let sanitized = text;

  // Remove XML/HTML tags that could break prompt structure
  sanitized = sanitized.replace(/<\/?[^>]+(>|$)/g, "");

  // Replace suspicious patterns with [REDACTED]
  for (const pattern of SUSPICIOUS_PATTERNS) {
    if (pattern.test(sanitized)) {
      console.warn(
        "[SECURITY] Suspicious pattern detected in transcript:",
        sanitized.match(pattern)?.[0],
      );
      sanitized = sanitized.replace(pattern, "[REDACTED - POLICY VIOLATION]");
    }
  }

  // Truncate to reasonable length (prevent token exhaustion attacks)
  const MAX_TRANSCRIPT_LENGTH = 500_000; // 500K chars ≈ 80K words
  if (sanitized.length > MAX_TRANSCRIPT_LENGTH) {
    console.warn(
      "[SECURITY] Transcript truncated from",
      sanitized.length,
      "to",
      MAX_TRANSCRIPT_LENGTH,
    );
    sanitized = sanitized.slice(0, MAX_TRANSCRIPT_LENGTH) + "\n[TRUNCATED]";
  }

  return sanitized;
}

export function sanitizeMeetingTitle(title: string): string {
  // Strip HTML, limit length, remove special chars
  return title
    .replace(/<\/?[^>]+(>|$)/g, "")
    .slice(0, 200)
    .trim();
}
```

#### 3.3 Add Request Size Limits

**File:** `src/server/api/middleware/request-limits.ts` (NEW)

```typescript
import type { MiddlewareHandler } from "hono";
import { HTTPException } from "hono/http-exception";
import type { AppBindings } from "../types";

const MAX_JSON_SIZE = 10 * 1024 * 1024; // 10MB
const MAX_HEADER_SIZE = 8 * 1024; // 8KB (CloudFlare limit is 16KB)

export const requestLimits: MiddlewareHandler<AppBindings> = async (c, next) => {
  // Check Content-Length header
  const contentLength = c.req.header("content-length");
  if (contentLength && parseInt(contentLength) > MAX_JSON_SIZE) {
    throw new HTTPException(413, {
      message: `Request body too large. Max ${MAX_JSON_SIZE / 1024 / 1024}MB allowed.`,
    });
  }

  // Check individual header sizes
  for (const [key, value] of Object.entries(c.req.header())) {
    if (value && value.length > MAX_HEADER_SIZE) {
      throw new HTTPException(431, { message: `Header ${key} too large` });
    }
  }

  await next();
};
```
